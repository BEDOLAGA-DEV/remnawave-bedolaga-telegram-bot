## Task 2: Phase 1 — renumber upstream migrations 0076-0087 → 0101-0112

**Context:** Upstream linear chain `0075←0076←…←0087`. Our head = `0100`. Map: 0076→0101 (down 0075→0100), … 0087→0112 (down 0086→0111). The 12 files arrived via merge (added, new filenames, not conflicted). Our 0096-0100 untouched.

- [ ] **Step 1: Create `_renumber_migrations_357.py` at repo root**

```python
"""One-off: renumber merged upstream migrations 0076-0087 -> 0101-0112,
rechaining the first onto our head 0100. Run once after `git merge upstream/main --no-commit`."""
import re
from pathlib import Path

VERS = Path('migrations/alembic/versions')
MAP = {f'{76+i:04d}': f'{101+i:04d}' for i in range(12)}   # 0076->0101 ... 0087->0112
DOWN_MAP = dict(MAP)
DOWN_MAP['0075'] = '0100'   # first upstream migration chains onto our head


def rev_of(text):
    m = re.search(r"^revision(?:\s*:\s*[^=]+)?\s*=\s*'([^']+)'", text, re.M)
    return m.group(1) if m else None


def main():
    targets = []
    for f in VERS.glob('*.py'):
        text = f.read_text(encoding='utf-8')
        if rev_of(text) in MAP:
            targets.append((f, text, rev_of(text)))
    if len(targets) != 12:
        print(f'WARNING: expected 12 target files, found {len(targets)}')
    for f, text, rev in targets:
        new_rev = MAP[rev]
        text = re.sub(r"(^revision(?:\s*:\s*[^=]+)?\s*=\s*')" + re.escape(rev) + r"(')",
                      lambda m: m.group(1) + new_rev + m.group(2), text, flags=re.M)
        m = re.search(r"^down_revision(?:\s*:\s*[^=]+)?\s*=\s*'([^']+)'", text, re.M)
        if m:
            old_down = m.group(1)
            new_down = DOWN_MAP.get(old_down, old_down)
            text = re.sub(r"(^down_revision(?:\s*:\s*[^=]+)?\s*=\s*')" + re.escape(old_down) + r"(')",
                          lambda mm: mm.group(1) + new_down + mm.group(2), text, flags=re.M)
        new_name = re.sub(r'^\d{4}_', new_rev + '_', f.name)
        f.write_text(text, encoding='utf-8')
        f.rename(f.with_name(new_name))
        print(f'{f.name} (rev {rev}->{new_rev}) -> {new_name}')


if __name__ == '__main__':
    main()
```
